import Foundation
import CoreGraphics
import ImageIO

// usage: measure <png> <hex> [tol] ; prints estimated stroke width in px via 2*A/P per component, median over big components
let args = CommandLine.arguments
guard args.count >= 3, let src = CGImageSourceCreateWithURL(URL(fileURLWithPath: args[1]) as CFURL, nil),
      let img = CGImageSourceCreateImageAtIndex(src, 0, nil) else { print("bad"); exit(1) }
let hex = UInt32(args[2], radix: 16)!
let tol = args.count > 3 ? Int(args[3])! : 14
let w = img.width, h = img.height
var buf = [UInt8](repeating: 0, count: w * h * 4)
let cs = CGColorSpaceCreateDeviceRGB()
let ctx = CGContext(data: &buf, width: w, height: h, bitsPerComponent: 8, bytesPerRow: w * 4, space: cs,
                    bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue)!
ctx.draw(img, in: CGRect(x: 0, y: 0, width: w, height: h))
let tr = Int((hex >> 16) & 255), tg = Int((hex >> 8) & 255), tb = Int(hex & 255)
var mask = [Bool](repeating: false, count: w * h)
// ignore top 200px and bottom 500px chrome (UI overlays)
let y0 = Int(args.count > 4 ? args[4] : "260")!, y1 = h - Int(args.count > 5 ? args[5] : "700")!
for y in y0..<y1 { for x in 0..<w {
    let i = (y * w + x) * 4
    if abs(Int(buf[i]) - tr) <= tol && abs(Int(buf[i+1]) - tg) <= tol && abs(Int(buf[i+2]) - tb) <= tol { mask[y*w+x] = true }
}}
var seen = [Bool](repeating: false, count: w * h)
var results: [(Int, Double)] = []
var total = 0
var mins: [Double] = []
for y in y0..<y1 { for x in 0..<w where mask[y*w+x] && !seen[y*w+x] {
    var stack = [(x, y)]; seen[y*w+x] = true
    var area = 0, per = 0; var pts: [(Int,Int)] = []
    while let (cx, cy) = stack.popLast() {
        area += 1; if pts.count < 6000 { pts.append((cx,cy)) }
        var edge = false
        for (dx, dy) in [(1,0),(-1,0),(0,1),(0,-1)] {
            let nx = cx+dx, ny = cy+dy
            if nx < 0 || ny < y0 || nx >= w || ny >= y1 || !mask[ny*w+nx] { edge = true; continue }
        }
        if edge { per += 1 }
        for dx in -1...1 { for dy in -1...1 {
            let nx = cx+dx, ny = cy+dy
            if nx >= 0, ny >= y0, nx < w, ny < y1, mask[ny*w+nx], !seen[ny*w+nx] { seen[ny*w+nx] = true; stack.append((nx, ny)) }
        }}
    }
    total += area
    if area >= 60 {
        var minW = 1e9
        if area < 3000 {
            for a in stride(from: 0.0, to: 180.0, by: 3.0) {
                let r = a * Double.pi / 180; let c = cos(r), s = sin(r)
                var lo = 1e9, hi = -1e9
                for (px, py) in pts { let p = Double(px) * c + Double(py) * s; lo = min(lo, p); hi = max(hi, p) }
                minW = min(minW, hi - lo + 1)
            }
            mins.append(minW)
        }
        results.append((area, 2.0 * Double(area) / Double(max(per, 1))))
    }
}}
results.sort { $0.1 < $1.1 }
let ws = results.map { $0.1 }
let med = ws.isEmpty ? 0 : ws[ws.count/2]
mins.sort(); print("dashMinWidthMedianPx=\(mins.isEmpty ? 0 : mins[mins.count/2]) n=\(mins.count)")
print("pixels=\(total) comps=\(results.count) medianWidthPx=\(String(format: "%.2f", med)) (pt@3x=\(String(format: "%.2f", med/3)))  p25=\(ws.isEmpty ? 0 : ws[ws.count/4]) p75=\(ws.isEmpty ? 0 : ws[ws.count*3/4])")
